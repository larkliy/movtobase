    12. Сборка проекта и модули¶

    Зачем это нужно и в чем прикол¶

    Один файл main.zig хорош для hello world, но реальный проект состоит из модулей, тестов, зависимостей и разных режимов сборки. В C для этого нужен Make/CMake/Meson и отдельный линкер. В Zig build-система написана на самом Zig и лежит в build.zig.

    Главное отличие: build.zig - это не XML и не DSL с макросами, а обычный Zig-код, который выполняется на этапе сборки. Вы описываете граф шагов: собрать исполняемый файл, запустить тесты, сгенерировать C-модуль, установить артефакт. Компилятор сам разбирается с зависимостями и кэшем.

    Вторая тема - модули. Вместо #include и заголовочников у каждого модуля есть корневой файл, а импорт выглядит как @import("mathx"). В build.zig вы явно связываете имя импорта с модулем.

    Базовый синтаксис¶

    Минимальный проект:

    proj/
├── build.zig
├── build.zig.zon
└── src/
    ├── main.zig
    └── mathx.zig

    build.zig.zon описывает пакет:

    .{
    .name = .proj,
    .version = "0.1.0",
    .minimum_zig_version = "0.17.0",
    .paths = .{""},
    .fingerprint = 0x520c3c9019b7dcad,
}

    fingerprint обязателен в 0.17. Проще всего создать проект через zig init - он подставит корректное значение. Если добавляете поле вручную, используйте значение, которое предложит компилятор в тексте ошибки.

    build.zig:

    const std = @import("std");

pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});

    const mathx = b.addModule("mathx", .{
        .root_source_file = b.path("src/mathx.zig"),
        .target = target,
        .optimize = optimize,
    });

    const exe = b.addExecutable(.{
        .name = "proj",
        .root_module = b.createModule(.{
            .root_source_file = b.path("src/main.zig"),
            .target = target,
            .optimize = optimize,
            .imports = &.{
                .{ .name = "mathx", .module = mathx },
            },
        }),
    });
    b.installArtifact(exe);

    const run_cmd = b.addRunArtifact(exe);
    run_cmd.step.dependOn(b.getInstallStep());
    run_cmd.addPassthruArgs();
    const run_step = b.step("run", "Run the app");
    run_step.dependOn(&run_cmd.step);

    const tests = b.addTest(.{ .root_module = mathx });
    const run_tests = b.addRunArtifact(tests);
    const test_step = b.step("test", "Run tests");
    test_step.dependOn(&run_tests.step);
}

    src/mathx.zig:

    pub fn square(x: i32) i32 {
    return x * x;
}

test "square" {
    const std = @import("std");
    try std.testing.expectEqual(@as(i32, 9), square(3));
}

    src/main.zig:

    const std = @import("std");
const mathx = @import("mathx");

pub fn main(init: std.process.Init) !void {
    var buffer: [1024]u8 = undefined;
    var out = std.Io.File.stdout().writerStreaming(init.io, &buffer);
    try out.interface.print("square(12)={d}\n", .{mathx.square(12)});
    try out.flush();
}

    Команды:

    zig build          # собрать и установить в zig-out/
zig build run      # запустить приложение
zig build test     # запустить тесты
zig build -Doptimize=ReleaseFast
zig build -Dtarget=x86_64-linux-gnu

    Важно: run_cmd.addPassthruArgs() в 0.17 заменил старый if (b.args) |args| run_cmd.addArgs(args). Поле b.args удалено, поэтому старые примеры build.zig нужно править.

    Модули и импорты¶

    b.addModule делает модуль публичным для других пакетов, b.createModule - приватным. Исполняемый файл получает root_module, в который передается список imports. Имя в imports - это ровно то, что вы напишете в @import("...").

    Модуль - не файл, а корневой файл плюс настройки. Все pub-декларации из корневого файла видны снаружи. Если модуль состоит из нескольких файлов, корневой файл реэкспортирует нужное:

    pub const parser = @import("parser.zig");
pub const lexer = @import("lexer.zig");

    Зависимости¶

    Пакеты добавляются в build.zig.zon:

    .dependencies = .{
    .known_folders = .{
        .url = "https://github.com/ziglibs/known-folders/archive/<commit>.tar.gz",
        .hash = "known_folders-0.0.0-...",
    },
},

    Хэш можно получить через zig fetch --save <url>. Затем модуль зависимости добавляется в imports через b.dependency(...).module(...). Точный API зависит от пакета, но идея всегда одна: зависимость - это модуль, а не набор заголовков и .lib-файлов.

    C-модули¶

    В 0.17 @cImport удален. Для C-заголовков используется b.addTranslateC:

    const translate = b.addTranslateC(.{
    .root_source_file = b.path("src/demo.h"),
    .target = target,
    .optimize = optimize,
});
const c_module = translate.createModule();

const exe = b.addExecutable(.{
    .name = "app",
    .root_module = b.createModule(.{
        .root_source_file = b.path("src/main.zig"),
        .target = target,
        .optimize = optimize,
        .imports = &.{
            .{ .name = "c", .module = c_module },
        },
    }),
});
exe.root_module.addCSourceFile(.{ .file = b.path("src/demo.c") });
exe.root_module.link_libc = true;

    Смотрим в сурсы: что там под капотом¶

    build.zig - это обычная программа, которая получает *std.Build и строит граф. Определения лежат в lib/std/Build.zig:

    pub fn addExecutable(b: *Build, options: ExecutableOptions) *Step.Compile
pub fn addModule(b: *Build, name: []const u8, options: Module.CreateOptions) *Module
pub fn createModule(b: *Build, options: Module.CreateOptions) *Module
pub fn addTest(b: *Build, options: TestOptions) *Step.Compile
pub fn addTranslateC(b: *Build, options: Step.TranslateC.Options) *Step.TranslateC

    b.path("src/main.zig") создает LazyPath - путь, который будет разрешен на этапе выполнения графа. Это важно для кэша: build-система отслеживает, какие файлы читает какой шаг.

    Сами шаги описаны в lib/std/Build/Step/. Например, Step.Compile - компиляция, Step.Run - запуск, Step.TranslateC - генерация Zig-кода из C-заголовка. Каждый шаг объявляет входы и выходы, поэтому повторный zig build не пересобирает то, что не изменилось.

    Посмотрите также lib/std/Build/Module.zig: модуль хранит root_source_file, target, optimize, imports, include_dirs, link_libc. Это и есть весь «конфиг» модуля.

    Где можно споткнуться¶

    1. build.zig.zon требует fingerprint. В 0.17 нельзя просто скопировать пример без fingerprint. Создайте проект через zig init или возьмите значение из ошибки компилятора.

    2. b.args больше нет. Старые примеры с if (b.args) |args| run_cmd.addArgs(args) не соберутся. Используйте run_cmd.addPassthruArgs().

    3. Импорт должен быть объявлен. @import("mathx") работает только если mathx есть в imports корневого модуля. Просто положить файл рядом недостаточно.

    4. addModule и createModule - разные вещи. Первый публикует модуль для потребителей пакета, второй создает приватный модуль для внутреннего использования. Для исполняемого файла обычно нужен createModule.

    5. Пути относительны корня проекта. b.path("src/main.zig") - от каталога с build.zig, а не от текущего каталога терминала. Не используйте std.fs.cwd() для путей сборки.

    6. build.zig выполняется на этапе сборки. В нем нельзя читать runtime-данные, зато можно генерировать файлы, запускать инструменты и строить граф. Ошибки в build.zig - это ошибки сборки, а не runtime.

    7. Кэш может скрывать проблемы. Если кажется, что изменения не применяются, проверьте, что файл действительно входит в граф. zig build --summary all показывает шаги.

    8. Зависимости с URL требуют хэш. Без хэша сборка не воспроизводима и Zig откажется. zig fetch --save обновляет build.zig.zon автоматически.
