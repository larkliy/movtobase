    23. Capstone: файловый индексатор¶

    Зачем это нужно и в чем прикол¶

    Это финальный проект курса. Он собирает вместе всё, что мы прошли:

    • std.process.Init и аргументы командной строки;
    • std.Io для вывода и работы с файловой системой;
    • рекурсивный обход каталога;
    • StringHashMap и ArrayList;
    • аллокаторы, defer и владение ключами;
    • сортировку и форматирование;
    • обработку ошибок и коды выхода;
    • build.zig, модули и тесты.

    Задача: утилита indexer <dir> считает файлы по расширениям и суммарный размер. Это не «hello world», а настоящая CLI-программа, которую можно запускать на своем проекте.

    Базовый синтаксис¶

    Структура проекта:

    indexer/
├── build.zig
├── build.zig.zon
└── src/
    ├── main.zig
    └── indexer.zig

    src/indexer.zig - вся логика. Она не зависит от main, поэтому ее легко тестировать:

    const std = @import("std");

pub const ExtCount = struct {
    ext: []const u8,
    files: usize,
    bytes: usize,
};

pub const Stats = struct {
    files: usize = 0,
    bytes: usize = 0,
    by_ext: std.StringHashMap(ExtCount),

    pub fn init(gpa: std.mem.Allocator) Stats {
        return .{ .by_ext = std.StringHashMap(ExtCount).init(gpa) };
    }

    pub fn deinit(self: *Stats, gpa: std.mem.Allocator) void {
        var keys = self.by_ext.keyIterator();
        while (keys.next()) |key| gpa.free(key.*);
        self.by_ext.deinit();
    }
};

fn shouldSkip(name: []const u8) bool {
    return std.mem.eql(u8, name, ".zig-cache") or
        std.mem.eql(u8, name, "zig-out") or
        std.mem.eql(u8, name, ".git");
}

pub fn extension(name: []const u8) []const u8 {
    const ext = std.fs.path.extension(name);
    return if (ext.len == 0) "<none>" else ext;
}

pub fn scan(io: std.Io, gpa: std.mem.Allocator, root: []const u8) !Stats {
    var stats = Stats.init(gpa);
    errdefer stats.deinit(gpa);

    var dir = try std.Io.Dir.cwd().openDir(io, root, .{ .iterate = true });
    defer dir.close(io);

    var walker = try dir.walk(gpa);
    defer walker.deinit();

    while (try walker.next(io)) |entry| {
        if (entry.kind == .directory) {
            if (shouldSkip(entry.basename)) walker.leave(io);
            continue;
        }
        if (entry.kind != .file) continue;

        const stat = try entry.dir.statFile(io, entry.basename, .{});
        const ext = extension(entry.basename);

        const key = try gpa.dupe(u8, ext);
        const gop = try stats.by_ext.getOrPut(key);
        if (gop.found_existing) {
            gpa.free(key);
        } else {
            gop.value_ptr.* = .{ .ext = key, .files = 0, .bytes = 0 };
        }
        gop.value_ptr.files += 1;
        gop.value_ptr.bytes += stat.size;

        stats.files += 1;
        stats.bytes += stat.size;
    }

    return stats;
}

pub fn lessThan(_: void, a: ExtCount, b: ExtCount) bool {
    if (a.bytes != b.bytes) return a.bytes > b.bytes;
    return std.mem.lessThan(u8, a.ext, b.ext);
}

test "extension" {
    try std.testing.expectEqualStrings(".zig", extension("main.zig"));
    try std.testing.expectEqualStrings("<none>", extension("Makefile"));
}

test "sorting" {
    var items = [_]ExtCount{
        .{ .ext = ".c", .files = 1, .bytes = 50 },
        .{ .ext = ".zig", .files = 2, .bytes = 100 },
    };
    std.mem.sort(ExtCount, &items, {}, lessThan);
    try std.testing.expectEqualStrings(".zig", items[0].ext);
}

    src/main.zig - CLI-обертка:

    const std = @import("std");
const indexer = @import("indexer");

pub fn main(init: std.process.Init) !u8 {
    var out_buf: [8192]u8 = undefined;
    var out = std.Io.File.stdout().writerStreaming(init.io, &out_buf);
    const gpa = init.gpa;

    var args = try std.process.Args.Iterator.initAllocator(init.minimal.args, gpa);
    defer args.deinit();
    _ = args.next();

    const root = args.next() orelse {
        try out.interface.print("usage: indexer <dir>\n", .{});
        try out.flush();
        return 2;
    };

    var stats = indexer.scan(init.io, gpa, root) catch |err| {
        try out.interface.print("scan failed: {s}\n", .{@errorName(err)});
        try out.flush();
        return 1;
    };
    defer stats.deinit(gpa);

    var list: std.ArrayList(indexer.ExtCount) = .empty;
    defer list.deinit(gpa);

    var it = stats.by_ext.iterator();
    while (it.next()) |kv| try list.append(gpa, kv.value_ptr.*);
    std.mem.sort(indexer.ExtCount, list.items, {}, indexer.lessThan);

    try out.interface.print("files: {d}, bytes: {d}\n", .{ stats.files, stats.bytes });
    for (list.items) |item| {
        try out.interface.print("{s}: files={d} bytes={d}\n", .{ item.ext, item.files, item.bytes });
    }
    try out.flush();
    return 0;
}

    build.zig:

    const std = @import("std");

pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});

    const indexer_module = b.addModule("indexer", .{
        .root_source_file = b.path("src/indexer.zig"),
        .target = target,
        .optimize = optimize,
    });

    const exe = b.addExecutable(.{
        .name = "indexer",
        .root_module = b.createModule(.{
            .root_source_file = b.path("src/main.zig"),
            .target = target,
            .optimize = optimize,
            .imports = &.{
                .{ .name = "indexer", .module = indexer_module },
            },
        }),
    });
    b.installArtifact(exe);

    const run_cmd = b.addRunArtifact(exe);
    run_cmd.step.dependOn(b.getInstallStep());
    run_cmd.addPassthruArgs();
    const run_step = b.step("run", "Run the indexer");
    run_step.dependOn(&run_cmd.step);

    const tests = b.addTest(.{ .root_module = indexer_module });
    const run_tests = b.addRunArtifact(tests);
    const test_step = b.step("test", "Run tests");
    test_step.dependOn(&run_tests.step);
}

    Смотрим в сурсы: что там под капотом¶

    std.Io.Dir.walk в lib/std/Io/Dir.zig возвращает Walker, который рекурсивно обходит каталог. Walker.next автоматически заходит в подкаталоги, а leave позволяет пропустить ненужное дерево. Walker.Entry содержит dir, basename, path, kind; dir - открытый каталог, в котором лежит запись, поэтому statFile можно делать без склейки длинного пути.

    std.StringHashMap в lib/std/hash_map.zig хранит ключи и значения в открытой адресации. Ключ не копируется: getOrPut доверяет тому, что вы передали. Поэтому в scan мы делаем dupe для расширения. Если getOrPut нашел существующий ключ, новый dupe освобождается.

    std.Io.Dir.statFile возвращает Stat с полем size. Для симлинков и специальных файлов поведение зависит от платформы; в проекте мы считаем только .file.

    std.process.Args.Iterator.initAllocator нужен на Windows, потому что аргументы приходят в WTF-16 и требуют преобразования. На POSIX можно было бы использовать init, но initAllocator переносим.

    Где можно споткнуться¶

    1. Ключи HashMap должны жить дольше таблицы. entry.basename указывает во внутренний буфер walker и меняется на следующей итерации. Поэтому dupe обязателен. Без него таблица будет содержать мусор.

    2. Освобождайте только новые ключи. В getOrPut при found_existing новый dupe уже не нужен. Если освободить существующий ключ, таблица сломается.

    3. Walker.next уже зашел в каталог. Чтобы пропустить .git, нужно вызвать walker.leave(io) сразу после получения directory entry. Иначе walker продолжит обход.

    4. entry.dir действителен до следующего next. Не сохраняйте его в долгоживущую структуру. Для статистики сохраняйте только данные (расширение, размер).

    5. statFile может упасть. Недостаточно прав, файл удалили, симлинк битый. В продакшене такие ошибки логируют и продолжают, а не падают всей программой. В учебной версии мы возвращаем ошибку наверх.

    6. Размер файла - не размер на диске. stat.size - логический размер. Для реального места на диске нужны блоки/кластеры, а это платформенно-зависимо.

    7. Расширение - не тип файла. .tar.gz даст .gz, Makefile - <none>. Если нужна более умная классификация, используйте MIME или список известных имен.

    8. Сортировка по убыванию. Компаратор возвращает true, если a должен идти раньше. Для убывания - a.bytes > b.bytes. Для стабильности при равных размерах добавлено сравнение по имени.

    9. Код выхода. 2 - неправильные аргументы, 1 - ошибка сканирования, 0 - успех. Это важно для shell-скриптов и CI.

    10. Release-сборка. В Debug индексер может быть заметно медленнее на больших деревьях. Для реального использования собирайте -Doptimize=ReleaseFast.
